function loadConfig() {
    if (fs.existsSync(configPath)) {

async function scanDirectory(dirPath, allSongs) {
    try {
        const files = fs.readdirSync(dirPath);
        for (const file of files) {
            const fullPath = path.join(dirPath, file);
            const stat = fs.statSync(fullPath);
            if (stat.isDirectory()) { await scanDirectory(fullPath, allSongs); }
            else if (/\.(mp3|wav|ogg|flac|m4a)$/i.test(file)) {
                try {
                    const metadata = await musicMetadata.parseFile(fullPath);
                    let art = null;
                    if (metadata.common.picture?.length) {
                        const pic = metadata.common.picture[0];
                        art = `data:${pic.format};base64,${pic.data.toString('base64')}`;
                    }
                    allSongs.push({ id: allSongs.length + 1, name: metadata.common.title || path.basename(file, path.extname(file)), artist: metadata.common.artist || "Unknown Artist", album: metadata.common.album || "Unknown Album", duration: metadata.format.duration || 0, file: fullPath, art, playCount: 0, playTime: 0 });
                } catch (err) {
                    allSongs.push({ id: allSongs.length + 1, name: path.basename(file, path.extname(file)), artist: "Unknown Artist", album: "Unknown Album", duration: 0, file: fullPath, art: null, playCount: 0, playTime: 0 });
                }
            }
        }
    } catch (e) { console.error("Failed to read directory:", e); }
}

async function uploadArtworkToWeb(base64Data) {
    if (!base64Data || uploadedArtCache[base64Data]) return uploadedArtCache[base64Data] || null;
    try {
        const base64ImageBytes = base64Data.split(',')[1];
        if (!base64ImageBytes) return null;
        const formData = new URLSearchParams(); formData.append('image', base64ImageBytes);
        const response = await net.fetch('https://api.imgbb.com/1/upload?key=6d207e02198a847aa98d0a2a901485a5', { method: 'POST', body: formData.toString(), headers: { 'Content-Type': 'application/x-www-form-urlencoded' } });
        const json = await response.json();
        if (json?.data?.url) { uploadedArtCache[base64Data] = json.data.url; return json.data.url; }
    } catch (err) { console.error("Failed uploading track artwork to RPC image host:", err); }
    return null;
}

ipcMain.on('update-presence', async (event, { title, artist, isPlaying, duration, currentTime, art }) => {
    if (!rpc) return;
    let targetImageKey = "prideicon";
    if (art?.startsWith('data:image')) { const remoteUrl = await uploadArtworkToWeb(art); if (remoteUrl) targetImageKey = remoteUrl; }
    const activityPayload = { type: 2, largeImageKey: targetImageKey, largeImageText: "by ivymroow >w< icon by nepvortex64 (sarah)", instance: false };
    if (isPlaying) {
        activityPayload.details = `${title}`; activityPayload.state = `by ${artist}`;
        if (duration && currentTime) activityPayload.endTimestamp = Math.floor(Date.now() / 1000 + Math.max(0, duration - currentTime));
    } else { activityPayload.details = `Paused: ${title}`; activityPayload.state = `by ${artist}`; }
    rpc.setActivity(activityPayload).catch(console.error);
});

ipcMain.handle('convert-download', async (event, { url, quality, playlist }) => {
    const dlPath = path.join(app.getPath('userData'), 'songs');
    if (!fs.existsSync(dlPath)) fs.mkdirSync(dlPath, { recursive: true });
    const oldPath = path.join(path.dirname(app.getPath('exe')), 'songs');
    if (fs.existsSync(oldPath) && oldPath !== dlPath) {
        for (const f of fs.readdirSync(oldPath)) {
            if (f.endsWith('.mp3') && !fs.existsSync(path.join(dlPath, f))) fs.copyFileSync(path.join(oldPath, f), path.join(dlPath, f));
        }
    }
    const args = [url, '-x', '--audio-format', 'mp3', '--audio-quality', quality || '0', '--embed-thumbnail', '--embed-metadata', '--postprocessor-args', 'ffmpeg:-c:v mjpeg -q:v 5', '-o', path.join(dlPath, playlist ? '%(playlist_title)s/%(title)s.%(ext)s' : '%(title)s.%(ext)s'), '--progress', '--newline', '--no-mtime'];
    if (!playlist) args.push('--no-playlist');
    const bin = [path.join(process.resourcesPath, 'yt-dlp.exe'), path.join(__dirname, 'yt-dlp.exe')].find(f => fs.existsSync(f));
    return new Promise(resolve => {
        if (!bin) { resolve({ success: false, error: 'yt-dlp.exe not found' }); return; }
        const before = Date.now(); let stderrBuf = '';
        const proc = require('child_process').spawn(bin, args, { windowsHide: true, stdio: ['ignore', 'pipe', 'pipe'] });
        proc.stdout.on('data', data => { const p = data.toString().match(/(\d+\.?\d*)%/); if (p) mainWindow.webContents.send('convert-progress', Math.round(parseFloat(p[1]))); });
        proc.stderr.on('data', data => { stderrBuf += data; });
        proc.on('error', err => { resolve({ success: false, error: err.message }); });
        proc.on('close', async code => {
            if (code !== 0) { resolve({ success: false, error: (stderrBuf || 'exit code ' + code).slice(0, 500) }); return; }
            function findMp3(dir, since) {
                let results = [];
                try { const entries = fs.readdirSync(dir, { withFileTypes: true }); for (const e of entries) { const fp = path.join(dir, e.name); if (e.isDirectory()) results = results.concat(findMp3(fp, since)); else if (e.name.endsWith('.mp3') && fs.statSync(fp).birthtimeMs > since) results.push(fp); } } catch(e) {}
                return results;
            }
            const files = findMp3(dlPath, before - 2000).sort().slice(0, 100);
            if (!files.length) { resolve({ success: true, songs: [] }); return; }
            const songs = [];
            for (const fp of files) {
                try {
                    const meta = await musicMetadata.parseFile(fp);
                    let title = meta.common.title || path.basename(fp).replace(/\.mp3$/i, '');
                    let artist = meta.common.artist || 'Unknown Artist';
                    if (artist === 'Unknown Artist') {
                        const dash = title.match(/^(.+?)\s*[-–—]\s*(.+)/);
                        if (dash) { artist = dash[1].trim(); title = dash[2].trim(); }
                        else { const dir = path.basename(path.dirname(fp)); const dd = dir.match(/^(.+?)\s*[-–—]\s*(.+)/); if (dd) artist = dd[1].trim(); }
                    }
                    let art = null;
                    if (meta.common.picture?.length) { const p = meta.common.picture[0]; art = `data:${p.format};base64,${p.data.toString('base64')}`; }
                    songs.push({ name: title, artist, art, file: `data:audio/mpeg;base64,${fs.readFileSync(fp).toString('base64')}` });
                } catch(e) {
                    const fname = path.basename(fp).replace(/\.mp3$/i, '');
                    const dash = fname.match(/^(.+?)\s*[-–—]\s*(.+)/);
                    let title = dash ? dash[2].trim() : fname;
                    let artist = dash ? dash[1].trim() : 'Unknown Artist';
                    if (artist === 'Unknown Artist') { const dir = path.basename(path.dirname(fp)); const dd = dir.match(/^(.+?)\s*[-–—]\s*(.+)/); if (dd) artist = dd[1].trim(); }
                    songs.push({ name: title, artist, art: null, file: `data:audio/mpeg;base64,${fs.readFileSync(fp).toString('base64')}` });
                }
            }
            resolve({ success: true, songs });
        });
    });
});

KEY JS FUNCTIONS:

--- Audio ---
let audio = new Audio();
togglePlay(), playS(song) - play song from IndexedDB
syncCenter() - position gallery cards

--- Library ---
renderUI(full, sync) - rebuild sidebar + gallery from DB
delSong(id) - instant DOM delete + async DB delete
clearLib() - reset all data

--- Search/Favorites ---
applyFilters() - DOM hide/show based on search + fav filter
toggleFavorite(id) - toggle heart icon

--- Playlist ---
renderPlaylists() - render playlist cards with context menu
openPlaylistDetail(pl) - view playlist tracks
ctxEditSongs() - edit playlist tracks/name via modal
savePlaylist() / saveEditedPlaylist() - create/edit

--- Social (P2P via PeerJS) ---
initPeer() - connect to 0.peerjs.com with custom ID
connectToPeer(remoteId) - direct P2P connection
handleSocialMessage(fromPeer, data) - message router
broadcastActivity() - send current track to connected friends
addSocialFriend() / removeSocialFriend() - friend management

--- Listen Together (P2P MP3 Streaming) ---
toggleListenTogether(peerId) - start/stop listen mode
sendSongChunks(peerId, songId) - chunk MP3 into 48KB pieces over data channel
playListenSong(data) - reassemble chunks into Blob URL, play
startListenSync(peerId) - 3s interval for position sync + track change detection

Listen together data flow:
1. User clicks "listen together" -> sends listen-req
2. Host responds with listen-start (total chunks) + listen-chunk messages
3. Receiver accumulates chunks, when complete -> Blob URL -> audio.src = URL
4. Host sends listen-sync every 3s with position + track_name
5. If host's track changes, host auto-sends new chunks for new song
6. Listen mode stays active until user clicks "listen together" again

--- Convert (yt-dlp) ---
startConvert() - shows overlay, calls convertDownload IPC
onConvertProgress(pct) - updates overlay percentage
Metadata parsing with parent-dir fallback for Bandcamp artist

== APPLY ALL FILTERS ===
applyFilters() function:
  1. Reads search input query
  2. Filters songs by query (name/artist/album)
  3. Filters by currentFilter ('all' or 'fav')
  4. Hides non-matching .track-item DOM elements
  5. Updates empty-notice visibility

=== MARQUEE ===
recalculateMarquee() - measures scrollWidth vs clientWidth
Sets titleMax = (textW - wrapW) + 40, capped for animation
Scrolls at 1px/frame, pauses 100 frames at each edge

=== GALLERY SYNCING ===
syncCenter() - positions .album-card elements:
  Active card: center, scale(1.1), full opacity
  Adjacent cards: scale(0.75), rotateY(±35deg), 0.3 opacity
  Far cards: hidden

=== AUTO-UPDATE ===
electron-updater with GitHub API provider
setTimeout(() => autoUpdater.checkForUpdatesAndNotify(), 3000)

=== IMPORTANT CSS VARIABLES ===
--reactive: changes based on current song's album art color
--neon / --brand: user's chosen theme color (default #57fe59)
--bg: #0a0a0a
--card-w: 320px (gallery card width)
